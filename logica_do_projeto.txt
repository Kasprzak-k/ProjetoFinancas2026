' --------------------------------------------------------
' CAMADA DE CONFIGURAÇÃO E ENTRADA
' --------------------------------------------------------
package "Entrypoint & Config" {
  class PortfolioApplication {
    + main(String[] args)
  }
  note right of PortfolioApplication
    **PortfolioApplication**
    A classe principal que inicializa o Spring Boot.
    Sobe o servidor embutido (Tomcat) e configura o contexto.
  end note

  class DataSeeder {
    + run(String... args)
  }
  note right of DataSeeder
    **DataSeeder (CommandLineRunner)**
    Roda ao iniciar o sistema. 
    Insere dados iniciais no PostgreSQL (mock de usuários e ativos) 
    para facilitar os testes locais.
  end note

  class RedisConfig {
    + redisTemplate()
  }
  note right of RedisConfig
    **RedisConfig**
    Configura a conexão com o Redis. 
    Necessário para o sistema de cache de portfólio.
  end note
}

' --------------------------------------------------------
' CAMADA DE CONTROLLERS (REST API)
' --------------------------------------------------------
package "REST API (Controllers)" {
  class PortfolioController {
    + getPortfolio(Long userId): PortfolioResponse
  }
  note top of PortfolioController
    **PortfolioController**
    Ponto de entrada para consultas financeiras.
    Recebe requisições HTTP GET para exibir a 
    posição completa e o lucro/prejuízo (PnL).
  end note

  class TransactionController {
    + createTransaction(TransactionRequest): TransactionResponse
  }
  note top of TransactionController
    **TransactionController**
    Ponto de entrada para registro de operações.
    Recebe requisições HTTP POST para efetivar 
    uma compra (BUY) ou venda (SELL) de um ativo.
  end note
}

' --------------------------------------------------------
' CAMADA DE SERVIÇOS (BUSINESS LOGIC)
' --------------------------------------------------------
package "Business Logic (Services)" {
  class PortfolioService {
    + getPortfolio(Long userId): PortfolioResponse
    - calculateAveragePrice()
    - calculatePnL()
  }
  note right of PortfolioService
    **PortfolioService**
    O "Cérebro" da leitura da carteira:
    1. Busca o histórico de transações.
    2. Calcula a quantidade atual de cada ativo.
    3. Calcula o preço médio pago.
    4. Consulta o PriceService para a cotação atual.
    5. Retorna o PnL (Lucro/Prejuízo).
    *Utiliza Redis para fazer cache desta resposta pesada.
  end note

  class TransactionService {
    + processTransaction(TransactionRequest): Transaction
    - validateSufficientPosition()
    - invalidatePortfolioCache(Long userId)
  }
  note right of TransactionService
    **TransactionService**
    O "Cérebro" das operações:
    1. Valida se o usuário e o ativo existem.
    2. Se for Venda (SELL), verifica se há saldo suficiente.
    3. Salva a transação no banco.
    4. Limpa o cache do Redis para que o portfólio seja recalculado.
  end note

  class PriceService {
    + getCurrentPrice(String ticker): BigDecimal
  }
  note bottom of PriceService
    **PriceService**
    Serviço auxiliar que simula ou 
    busca a cotação em tempo real 
    dos ativos no mercado.
  end note
}

' --------------------------------------------------------
' CAMADA DE DOMÍNIO (ENTIDADES E REPOSITÓRIOS)
' --------------------------------------------------------
package "Data Access & Domain" {
  
  interface UserRepository {
    + findById(Long id)
  }
  class User {
    - id: Long
    - name: String
    - email: String
  }

  interface AssetRepository {
    + findByTicker(String ticker)
  }
  class Asset {
    - id: Long
    - ticker: String
    - type: AssetType
  }

  interface TransactionRepository {
    + findByUserId(Long userId)
    + save(Transaction)
  }
  class Transaction {
    - id: Long
    - user: User
    - asset: Asset
    - type: TransactionType
    - quantity: BigDecimal
    - price: BigDecimal
    - date: LocalDateTime
  }

  note bottom of Transaction
    As entidades (User, Asset, Transaction) mapeiam
    diretamente para as tabelas do PostgreSQL via Hibernate.
    Os Repositórios (Interfaces Spring Data) geram as 
    queries SQL automaticamente (SELECT, INSERT, etc).
  end note
}

' --------------------------------------------------------

TransactionController --> TransactionService : "Chama p/ registrar"

PortfolioService --> TransactionRepository : "Lê histórico"
PortfolioService --> PriceService : "Lê cotação"

TransactionService --> UserRepository : "Valida"
TransactionService --> AssetRepository : "Valida"
TransactionService --> TransactionRepository : "Persiste operação"
